%============================================================ 2. the job ======
\begin{frame}{The job, and the one rule behind it}
  \begin{columns}[T]
    % \footnotesize, not \small, in both columns: at \small this frame
    % overflowed by 10.2pt (measured in the log, not guessed).  Nine lines a
    % column x the ~1.5pt of leading between the two sizes is ~14pt, so
    % stepping the size down once clears it without cutting a word.
    \begin{column}{0.5\linewidth}
      \footnotesize
      \textbf{What a PM asked for}
      \begin{itemize}
        \item Tell me when retail is getting \emph{euphoric} about something
              I hold.
        \item Tell me when that euphoria is \emph{fading} --- before the price
              tells me.
        \item Speak \textbf{rarely}. A light that is always on is not a light.
      \end{itemize}
      \vspace{0.5em}
      \textbf{So success was defined before any test was run:} an alert inside
      $[\text{peak}-30\text{d},\ \text{peak}+1\text{d}]$ of a genuine top,
      where a top is a run of $\geq$25\% (ETF) / $\geq$50\% (single) followed
      by a $\geq$15\% / $\geq$30\% drawdown within 90 days.
    \end{column}
    \begin{column}{0.46\linewidth}
      \footnotesize
      \textbf{The one rule the project is built on}
      \vspace{0.3em}

      Every number in this system is one of:
      \begin{itemize}
        \item \textbf{learned} from past data only, then frozen;
        \item \textbf{derived} from a stated budget or definition;
        \item a \textbf{convention}, published with the ablation that shows
              what it costs;
        \item a \textbf{desk decision}, named as one.
      \end{itemize}
      \vspace{0.4em}
      {\color{gicgrey}There is no fourth category. ``It looked about right''
       is not in the register.}
    \end{column}
  \end{columns}
  \sowhat{The deliverable is a \emph{risk} instrument --- a warning light for
    crowded longs --- not an alpha signal. Everything after this slide is
    tested against that claim and nothing wider.}
\end{frame}

%=================================================== 3. what it measures ======
\begin{frame}{What is actually measured --- and what never is}
  \begin{columns}[T]
    \begin{column}{0.52\linewidth}
      \small
      \textbf{Euphoria} = the crowd has stopped analysing a name and started
      celebrating it. Four crowd measurements, each ranked against
      \emph{that name's own} trailing year:
      \begin{enumerate}\small
        \item \textbf{Attention extremity} --- you cannot be euphoric quietly.
        \item \textbf{Sustained bullishness} --- weeks of one-way lean, not one
              loud afternoon.
        \item \textbf{Crowd influx} --- new people still arriving.
        \item \textbf{Super-exponential growth} --- attention whose
              \emph{growth rate} is itself growing.
      \end{enumerate}
      \vspace{0.3em}
      {\scriptsize (4) is Sornette's LPPLS bubble signature applied to the
       \emph{crowd} instead of the chart: attention spreads as an epidemic, and
       a contagion that accelerates must saturate.}
    \end{column}
    \begin{column}{0.44\linewidth}
      \small
      \textbf{Per-name ranking is the whole trick.} ``Extreme'' always means
      extreme \emph{for this name}, so BTC is judged against loud-for-BTC and a
      utility ETF against quiet-for-utilities. One global threshold would fire
      on BTC every week and never on anything else.
      \vspace{0.6em}

      \textbf{What never enters the prediction:} the price chart. Price
      \emph{gates} an ending (you cannot end a party that never started) and
      \emph{grades} every alert --- but the crowd does the predicting.
      \vspace{0.4em}

      {\color{gicgrey}\scriptsize That restriction costs measured capture. It
       is kept because it is the only way to claim: \emph{the crowd alone
       called the top}.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The evidence base}
  % Four counters across the slide is 93.6pt each, so every label wraps to two
  % or three \scriptsize lines and the TALLEST label sets the height of the
  % whole row.  This frame overflowed by 19.1pt; the fix is two lines of label
  % (Bloomberg moved into the walk-forward bullet, where it belongs anyway)
  % plus the two vertical struts, which together are ~24pt.
  \stat{9 years}{of daily Reddit attention, 2017--2026}\hfill
  \stat{333}{hype episodes labelled before any detector existed}\hfill
  \stat{122}{tops the detector could possibly have caught}\hfill
  \stat{12{,}528}{authors scored on their own forecasting record}
  \vspace{0.9em}

  \begin{itemize}\footnotesize
    \item \textbf{The exam was written before it was sat.} Episodes, tops and
      the detectable subset are defined in Notebook 01 by a rule applied to
      price alone --- with no detector in existence and no feature consulted.
      Nothing downstream can move the target.
    \item \textbf{Two denominators are quoted everywhere.} 24 of 122
      \emph{detectable} tops is also 7.2\% of all 333 episodes. Crowd coverage
      is deep in 2020--22 and 2026, thin to absent in 2023--25: there the
      detector is \textbf{blind}, which is not the same as wrong.
    \item \textbf{Walk-forward throughout, against real closes.} Prices are
      Bloomberg; every threshold is fitted on \emph{earlier years only} and
      then frozen, and the live pipeline snapshots each day's signal and never
      revises it.
  \end{itemize}
  \sowhat{The honest headline is not a hit rate --- it is that the target,
    the split and the budget were all fixed in advance, so the hit rate means
    something.}
\end{frame}

%================================================== 5. the danger state =======
\begin{frame}{The strongest result: the danger state}
  \begin{columns}[T]
    \begin{column}{0.52\linewidth}
      \vspace{0.4em}
      \stat[0.95]{62\%}{of danger-state days are followed by a
        $\geq$10\%-in-7-days drop within 30 days}\\[1.0em]
      \stat[0.95]{19\%}{on ordinary candidate days}\\[1.0em]
      {\scriptsize 90\% instrument-cluster CI on the uplift:
       $[+28\text{pp}, +50\text{pp}]$}
    \end{column}
    \begin{column}{0.44\linewidth}
      \scriptsize
      \textbf{The danger state is not a model.} It is the conjunction of two
      things the system already computes:
      \begin{itemize}
        \item the crowd is $\geq 2\times$ its own 120-day normal, \emph{and}
        \item price is already in a boom off its own 120-day low.
      \end{itemize}
      \vspace{0.5em}
      A combined price+crowd \emph{alert} model was also tested and did
      \textbf{not} clear its criterion: its 65\% hit rate is indistinguishable
      from the 62\% baseline of its own candidate days.
      \vspace{0.4em}
      \\ \textbf{Which is itself the finding:} the \emph{state} carries the
      information, so the product ships it as a standing amber band and uses
      the END alerts to time the peak \emph{within} it.
    \end{column}
  \end{columns}
  \sowhat{The most useful thing this system produces is a \emph{regime flag}
    that roughly triples the base rate of a near-term double-digit drawdown.}
\end{frame}

\begin{frame}{The walk-forward operating record}
  \begin{table}
    \scriptsize
    \setlength{\tabcolsep}{4pt}
    \begin{tabular}{@{}lrrrrr@{}}
      \toprule
      & \textbf{Captured} & \textbf{Detectable} & \textbf{Rate}
      & \textbf{False alarms} & \textbf{FA / instr-yr} \\
      \midrule
      GET OUT (shipped: boom-gated, smoothed) & 24 & 122 & 19.7\% & 39 & 0.195 \\
      GET IN  (shipped: phase-aware, smoothed) & 20 & 125 & 16.0\% & 124 & 0.25 \\
      \bottomrule
    \end{tabular}
  \end{table}
  \vspace{0.2em}
  \begin{itemize}\footnotesize
    \item \textbf{Lead time.} GET OUT fires a median \textbf{8 days before the
      peak}; GET IN a median 17 days after the trough, with a median
      \textbf{66 days of rally still ahead}.
    \item \textbf{Read those two differently.} 66 days of rally ahead is
      \emph{actionable}. Eight days before a top is not time to build a short
      --- it is time to \emph{stop adding}. The deck says so rather than
      rounding it up into an entry signal.
    \item \textbf{One number is above budget, and it is printed on the pane
      itself:} GET IN false alarms run 0.25/instrument-year against a derived
      budget of 0.23.
  \end{itemize}
  \sowhat{One in five detectable tops, at one false alarm per five
    instrument-years, with eight days' warning. Useful as a risk flag;
    nowhere near a standalone strategy --- and priced as such.}
\end{frame}

%================================================== 8. the blip fix ===========
\begin{frame}{The desk complaint, fixed at the signal and not with a rule}
  \begin{columns}[T]
    % Worst overflow in the deck at 42.2pt, and the arithmetic says why: the
    % left column ran ~11 \small lines plus a 5-line list (~180pt) into a body
    % that is only ~140pt once the SO WHAT strip has taken its ~42pt.  Both
    % type sizes step down once and the two struts halve; nothing is cut.  The
    % right column is trimmed too, because in a columns[T] the frame takes the
    % MAX of the two -- shrinking only the taller one just hands the overflow
    % to its neighbour.
    \begin{column}{0.5\linewidth}
      \footnotesize
      \textbf{The complaint:} ``euphoria for a single day, then gone'' --- and
      START markers landing next to END markers.
      \vspace{0.3em}

      \textbf{The tempting fix:} a minimum-duration rule. Rejected --- it hides
      the symptom and adds a constant defensible only by taste.
      \vspace{0.3em}

      \textbf{What was done instead:} two changes to the signal, each measured.
      \begin{itemize}\scriptsize
        \item Trigger on the \textbf{7-day-smoothed} score. Quality
          AP $0.435 \rightarrow 0.449$, at a recorded cost of 2 captures.
        \item Make GET IN \textbf{phase-aware}: a day that already meets every
          ending condition is end-stage, so a ``start'' there cannot fire.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\linewidth}
      \stat[0.95]{20 $\rightarrow$ 2}{START markers printing adjacent to an
        END, before and after}\\[0.7em]
      \stat[0.95]{29 $\rightarrow$ 20}{captures --- the price paid, recorded
        rather than buried}
      \vspace{0.2em}

      {\scriptsize\color{gicgrey} Why pay it: a START landing on an END
       destroys trust in the whole pane, and an untrusted pane has an effective
       hit rate of zero.}
    \end{column}
  \end{columns}
  \sowhat{This is the methodological claim of the project in one slide: fix the
    \emph{signal}, publish what the fix cost, and never patch the output with a
    rule you cannot derive.}
\end{frame}

%=============================================== 10. rejected trades ==========
\begin{frame}{What was tested and \textbf{rejected}}
  \small
  \begin{table}
    \begin{tabular}{@{}lrrrrl@{}}
      \toprule
      \textbf{Claim} & \textbf{n} & \textbf{Alert 20d} & \textbf{Baseline}
      & \textbf{Diff} & \textbf{90\% CI} \\
      \midrule
      GET IN $\rightarrow$ BUY   & 215 & $+0.34\%$ & $+0.98\%$ & $-0.63\%$
        & $[-1.51\%, +0.26\%]$ \\
      GET OUT $\rightarrow$ SELL & 67  & $-0.21\%$ & $+1.34\%$ & $-1.55\%$
        & $[-5.63\%, +2.63\%]$ \\
      \bottomrule
    \end{tabular}
  \end{table}
  % 1.2pt over: the strut that used to sit here was worth 2.2pt, which is the
  % whole debt plus a hair.  Nothing else on the slide moves.
  \begin{itemize}\footnotesize
    \item \textbf{Criterion, written before any return was computed:}
      $n \geq 30$ judged alerts; mean 20-day forward return beats the
      same-instrument unconditional baseline in the expected direction; 90\%
      instrument-cluster CI of the difference excludes zero.
    \item \textbf{Both REJECTED.} The SELL direction is right, but at $n=67$
      the interval spans zero. The BUY edge is simply not there at 20 days. A
      short-after-GET-OUT overlay was also measured: annualised Sharpe
      $-0.82$.
    \item \textbf{A gradient-boosted model} on the same features caught 11
      endings against the shipped rules' 23, inside the same tournament ---
      and the weights it \emph{learned} rank the same features top, which is
      independent evidence the rules are not arbitrary.
  \end{itemize}
  \sowhat{Avoiding the left tail of a long you already hold is a different and
    easier problem than timing a short entry. The evidence supports the first
    claim only, so that is the only claim made.}
\end{frame}

%=============================================== 12. the null result =========
\begin{frame}{The influence result we are \emph{not} allowed to claim}
  \sidebal{0.54}{0.44}
  \sidefig{05/10_only-rung-0-clears-zero-the-graph-does-not-ear.png}{%
    The adoption ladder. Each rung is a more complex model against the one
    below it: 10 paired seeds, cluster-bootstrapped 90\% interval on the AP
    difference, Bonferroni-corrected across six comparisons.}{%
    Eight graph architectures were tested --- GraphSAGE, GCN, SGC/MixHop,
    H2GCN, label propagation, an MLP. \textbf{Only rung 0 clears zero:} a plain
    linear model beats random, and \emph{nothing beats the plain linear model.}
    \vspace{0.4em}
    \\ Our AP of \textbf{0.098} against a 0.047 random floor is in the same
    place as the thesis's own GraphSAGE result (0.140 on its data), so this is
    a replication of a small effect --- not a failure to find a large one.
    \vspace{0.3em}
    \\ \textbf{So the product ships no model here.} It shows the measured
    record only.}
\end{frame}
