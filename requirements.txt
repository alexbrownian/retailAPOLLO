# retailAPOLLO dependencies.
# Install with the same interpreter that runs the pipeline:
#   pip install -r requirements.txt --user

# Core data stack.
pandas
pyarrow
numpy
scipy

# Ingestion.
zstandard          # Streamed (de)compression of raw post files.
requests
huggingface_hub    # Historical X dataset downloads (external machine).
fsspec

# Sentiment.
vaderSentiment
finvader           # Optional financial lexicons; degrades gracefully if absent.
joblib             # Parallel scoring; also scikit-learn's parallel backend.
                   # Install and upgrade together with scikit-learn: a
                   # mismatched pair emits high-volume UserWarnings from
                   # sklearn's fit paths (cosmetic, filtered narrowly in
                   # analytics/retail_flow.py).

# Extraction.
wordfreq           # Word-vs-ticker screening fallback.

# Modelling. A live-pipeline dependency: the desk GET IN / GET OUT pair
# (analytics/ml_detector.py) and the retail-flow dial
# (analytics/retail_flow.py) run inside `--what phases`. All sklearn
# imports are function-level, so a dashboard-only machine may omit it.
scikit-learn

# Notebook re-execution (research machines only). Invoke as
# `python -m jupyter nbconvert ...`: on Windows the Scripts folder
# holding jupyter.exe is often not on PATH, and `python -m` resolves the
# module through the interpreter instead.
jupyter
nbconvert
jupytext
matplotlib         # Notebook figures (the dashboard uses plotly).

# Dashboard.
streamlit
plotly

# Tests.
pytest

# Windows corporate-proxy TLS (harmless elsewhere).
pip-system-certs
