# retailAPOLLO workstation extras.
#
# Everything a research/operator machine needs ON TOP of
# requirements.txt. Kept separate because Streamlit Community Cloud
# installs requirements.txt automatically, and none of the packages
# below belong on the hosted dashboard: it re-executes no notebooks,
# runs no tests, downloads no vendor datasets, and is not behind a
# Windows corporate proxy. Every entry here is install weight and
# failure surface the host does not need.
#
#   pip install -r requirements.txt -r tools/requirements-dev.txt --user
#
# (See requirements.txt for the Windows --user note.)

# Notebook re-execution (research machines only). Invoke as
# `python -m jupyter nbconvert ...`: on Windows the Scripts folder
# holding jupyter.exe is often not on PATH, and `python -m` resolves the
# module through the interpreter instead.
jupyter
nbconvert
jupytext
matplotlib         # Notebook figures (the dashboard uses plotly).

# Historical X dataset downloads.
huggingface_hub

# Tests.
pytest

# Windows corporate-proxy TLS. Windows-only by design: it patches the
# certificate store at interpreter startup. It also tries to touch the
# pip installation itself, which fails outright on a Linux container
# whose pip is distribution-managed:
#
#   ERROR: Cannot uninstall pip <ver>, RECORD file not found.
#
# That is why it must never reach the hosted requirements file.
pip-system-certs

# blpapi (Bloomberg, pull_bloomberg_prices.py) is not on PyPI. Install
# from Bloomberg's index with the Terminal running:
#   python -m pip install --index-url=https://blpapi.bloomberg.com/repository/releases/python/simple/ blpapi --user
