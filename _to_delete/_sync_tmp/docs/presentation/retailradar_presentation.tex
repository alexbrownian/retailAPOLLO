% Falls back to a visible placeholder if the PNG has not been exported yet,
% so an incomplete export never blocks a rehearsal.
%
% The height cap is a MACRO, not a literal, for one practical reason: a slide
% carrying a figure plus a caption plus a SO WHAT strip overflows the frame,
% and the fix is always "give the picture less height", never "delete the
% reasoning".  A frame that needs it says \tightfig (or \tighterfig) at the
% top; because a frame is a TeX group the setting reverts on \end{frame}, so
% one slide's squeeze cannot leak into the next.
% \figh{0.41} sets the cap to an exact measured value; \tightfig and
% \tighterfig are just the two named presets.  The exact values in this file
% are not taste -- they were FITTED: compile, read the "Overfull \vbox (Xpt too
% high)" figure out of the log, and take exactly X pt (plus 4pt of slack) off
% the picture.  Which dimension to take it from depends on what governs:
%
%     rendered height  h = min(\figmaxh * \textheight,  width * L * aspect)
%
% so a TALL figure is cap-governed (lower \figh) and a WIDE one is
% width-governed (lower the \gfig width fraction -- lowering the cap on a wide
% figure does nothing at all, which is the trap this comment exists to record).
% Measured on this class: \textheight = 252.07pt, \linewidth = 398.34pt, and
% L = 199.17pt inside a half-width column.
\newcommand{\figmaxh}{0.62}
\newcommand{\figh}[1]{\renewcommand{\figmaxh}{#1}}
\newcommand{\tightfig}{\figh{0.46}}
\newcommand{\tighterfig}{\figh{0.38}}

    \end{center}

      {\scriptsize\color{gicgrey}#3\par}\fi}{%
    \begin{center}

% \sidefig{path}{caption}{so what} -- figure LEFT, words RIGHT.
%
% Why a second layout exists at all: on a 16:9 slide the text block is 398pt
% wide and 252pt tall, so its own aspect is 0.63.  A figure TALLER than that
% (a square heatmap at 0.81, a full-page screenshot at 0.69) is height-bound
% when stacked above its caption -- it hits the cap long before it runs out of
% width, and the slide ends up with a small picture floating between two wide
% empty margins.  Turning the slide sideways gives the picture the full text
% height and the words the leftover column, which roughly DOUBLES the rendered
% area of a screenshot with no words removed.  Wide figures (aspect < 0.6) are
% better off in the stacked layout and keep using \gfig directly.
% The split between the two columns is a knob, for the reason the figure height
% cap is one.  In a sideways layout there are TWO things that can be the tall
% one, and only measurement says which: on most frames it is the picture, but on
% a frame whose caption and SO WHAT together run ~28 lines the WORDS column
% governs and shrinking the picture does nothing.  Widening the words column by
% 15% removes three or four lines of wrap, which is 30-40pt -- and on the
% stacked-pair layout it costs the picture nothing at all, because there the
% picture is height-capped and never uses its full column width anyway.
% So: \sidebal{fig}{words} before the figure, on the frames that need it.
% It is local to the frame, like \figh, because a beamer frame is a group.
% Each layout has its OWN sensible default split (they differ -- see
% \sidefigtwo), so the two are kept apart: \sidebal records an explicit
% per-frame override and raises a flag; \sidedefault installs a layout's
% built-in split only when that flag is down.  Without the flag a layout
% default would silently overwrite the override written one line above it.

\section{Aims and hypothesis}
\begin{frame}{Aims --- what a PM actually needs}
  \begin{itemize}
    \item \textbf{The problem.} Retail attention episodes --- GameStop 2021,
      gold 2026, the meme complex --- follow a recognisable arc: a crowd
      arrives, sentiment goes one-way, attention grows super-exponentially,
      price peaks, and a bust of 15--90\% follows within weeks.
    \item \textbf{What the literature already gives us.} Attention extremes
      predict \emph{reversal}, not continuation (Barber \& Odean 2008; Barber
      et al.\ 2022). That is a correlation.
    \item \textbf{What a desk needs instead is a clock.} Has an episode
      \emph{started} (the rally has fuel left), and is it \emph{ending} (the
      top is near)? Two different questions, two different observable
      signatures.
  \end{itemize}
  \sowhat{The deliverable is not a return forecast. It is a two-state warning
    a PM can read in ten seconds: \textbf{\color{bullish}GET IN} --- a crowd
    episode is starting on this name; \textbf{\color{bearish}GET OUT} --- the
    one we are already in is ending.}
\end{frame}

\begin{frame}{The one rule the whole project is built on}
  \vspace{0.5em}
  \begin{beamercolorbox}[wd=\linewidth,sep=1em]{block body}
    \textbf{\color{gicnavy}The crowd-only rule.} Every \emph{predictive} input
    is built from social-media aggregates. Price never enters any feature,
    level, gate or alert in the headline detectors. Price exists only to
    define ground truth and to score the detectors.
    \\[0.6em]
    A unit test enforces it: no detector function accepts a price argument.
  \end{beamercolorbox}
  \vspace{0.8em}
  \begin{itemize}\small
    \item \textbf{Why bother.} The claim being defended is \emph{``the crowd
      alone called it''}. The moment price leaks into a feature, that claim is
      unfalsifiable and the study is worth nothing.
    \item \textbf{The recorded exception.} A \emph{second} signal family
      (2026-07-24 desk decision) permits price as a \textbf{gate} --- a
      prerequisite, never a score input. It is reported separately, and the
      crowd-only detectors remain the research headline.
  \end{itemize}
\end{frame}

\section{Results --- quantitative}
\begin{frame}{The walk-forward operating record}

      & \textbf{False alarms} & \textbf{FA / instr-yr} \\
      \midrule

  \vspace{0.4em}
  \begin{itemize}\footnotesize
    \item \textbf{Lead times.} GET OUT fires a median \textbf{8 days} before

      a median \textbf{66 days of rally still ahead}.
    \item \textbf{Two denominators, always.} 29 of 125 \emph{detectable}
      starts is also 8.7\% of all 333 episodes. Coverage is deep in
      2020--22 and 2026 and thin to absent in 2023--25 --- the detector is
      \emph{blind} there, which is not the same as wrong.

      false alarms run 0.348/instrument-year against a derived budget of 0.23.
  \end{itemize}

\section{Limitations and next}
\begin{frame}{Limitations --- pre-answered}
  \footnotesize
  \begin{enumerate}
    \item \textbf{Coverage deserts.} 2023--25 archive coverage is thin to
      zero. The detectors are blind there, not validated there.
    \item \textbf{Trough truncation.} The trough is a measured 120-day low, so
      run lengths cap at 120 days by construction.
    \item \textbf{V-recoveries qualify.} A boom off a 120-day low admits
      post-crash rebounds (energy 2020) as episodes --- inherited knowingly;
      the crowd features must \emph{earn} the mania-vs-recovery distinction.
    \item \textbf{Onset false alarms are above budget}: 0.348 vs 0.23 per
      instrument-year. Printed on the pane, not excused.
    \item \textbf{No standalone alpha claim.} Tested, rejected, recorded.
    \item \textbf{Calibration is not monotone} at low score deciles ---
      annotated on the figure, not smoothed away.
    \item \textbf{The influence model is transductive.} Its headline numbers
      assume the author was present when the model was fitted. On unseen
      authors it falls to the random floor, so it ranks a known board --- it
      does not screen newcomers.
    \item \textbf{Ground truth requires a confirmed bust.} Euphoria that
      deflates slowly never enters the catalog, so capture rates speak only to
      the arcs a desk most needs flagged.
  \end{enumerate}
\end{frame}

\begin{frame}{Future work}
  \begin{itemize}\small
    \item \textbf{Fold comments into the abstracted aggregates} --- the single
      largest capture-recovery lever for the coverage deserts, and the one
      change most likely to move 2023--25 from ``blind'' to ``measured''.
    \item \textbf{LLM-based call extraction} to replace the VADER stance
      heuristic, which is the weakest link in the influence study's labels.
    \item \textbf{Graph features as euphoria inputs} once the influence store
      passes its maturity bar.
    \item \textbf{Re-admit \texttt{source\_breadth}} once the multi-source
      archive spans a full episode cycle --- the rejection was about coverage
      regime, not about the idea.
    \item \textbf{Two watch items auto-re-test at the next research pass:}
      the logistic-weighted GET OUT (30 vs 24 captures, gain CI $[-1,+11]$)
      and per-kind thresholds (false alarms halved at equal capture rate, but
      on unmatched years).
  \end{itemize}
\end{frame}

\begin{frame}{Conclusion}
  \vspace{0.3em}
  \begin{itemize}
    \item Two rule-based, crowd-only detectors --- one for the start of retail
      euphoria, one for its end --- survive a pre-registered tournament
      against learned challengers.
    \item They post honest walk-forward records, \textbf{fail} an
      honestly-specified alpha test, and earn their place as \textbf{risk
      timing and monitoring infrastructure} rather than as a black-box signal.
    \item Every fitted number is either derived from an already-validated
      constant or walk-forward-learned from the past only. Every negative
      result is documented with the same care as the positives.
  \end{itemize}
  \vspace{0.8em}
  \begin{beamercolorbox}[wd=\linewidth,sep=0.9em]{block body}
    That --- more than any single capture rate --- is the claim this project
    defends.
  \end{beamercolorbox}
\end{frame}

    authors have almost no neighbourhood for a graph model to aggregate over.
    That is the structural reason the graph architectures fail later --- there
    is nothing to pass a message across for the typical node.}
\end{frame}

\begin{frame}{HONESTY --- centrality alone does not find the useful voices}

\begin{frame}{HONESTY --- what the model confuses, and why}

\begin{frame}{DIAGNOSTIC --- leave-one-out ablation, and where the twins are}
  \figh{0.310}
  \gfig{05/13_leave-one-out-ablation.png}{0.98}{%
    \emph{Left}: change in AP when each feature is removed --- left of zero

  \sowhat{Calls per active day, bearish share and directional entropy carry

\begin{frame}{DIAGNOSTIC --- the one bank change that cleared zero}
  \tightfig
\gfig{05/14_feature-bank-candidates-green-cleared-zero-at.png}{0.679}{%

    bank, 95\% intervals over 10 seeds. Green cleared zero; pink did not.}
  \sowhat{Only \emph{drop the behavioural family} cleared zero
    ($+0.0075$, $[+0.0022, +0.0128]$) --- and it then had to survive a
    Bonferroni correction across the six candidates tested in the same round,

    being an accident.}
\end{frame}

\begin{frame}{NULL --- when corruption \emph{improves} the model}
  \tightfig
\gfig{05/15_sage-lite-ap-under-corruption-dice-improves-it.png}{0.714}{%
    GraphSAGE-lite under three edge perturbations. \emph{Left}: AP. Under DICE

    destroyed. \emph{Right}: accuracy over the same sweep.}
  \sowhat{This is a diagnostic disguised as an absurdity. If deliberately
